# 判题排队机制设计方案

## 🔍 当前判题系统分析

### 原有HUSTOJ判题机制

**判题核心**：
- ✅ **已有排队机制**：OJ_WT0（提交排队）、OJ_WT1（重判排队）
- ✅ **分布式支持**：UDP通知、JudgeHub分布式判题
- ✅ **队列支持**：Redis队列可选

**判题流程**：
```
用户提交 → 数据库排队(OJ_WT0) → judged守护进程 → judge_client判题 → 结果入库
```

### 资源占用分析（2核2G服务器）

**单个判题任务占用**：
- CPU：编译+运行 ≈ 0.5-1核（30秒）
- 内存：编译器+程序 ≈ 50-100MB
- IO：文件读写 ≈ 中等

**并发限制**：
- **CPU**：2核，同时最多2个判题任务
- **内存**：2GB，最多支持10-20个并发
- **综合限制**：建议最大并发数 = 2

### 核心设计原则

1. **严格并发控制**：最大同时2个判题任务
2. **优先级队列**：比赛题目优先处理
3. **超时控制**：单个任务最长60秒
4. **资源监控**：CPU/内存实时监控
5. **队列可视化**：用户查看排队进度

### 2. **判题守护进程设计**

#### Next.js判题守护进程

```typescript
// 判题守护进程服务
class JudgeDaemonService {
  private maxConcurrent = 2;         // 最大并发数
  private runningCount = 0;          // 正在判题的数量
  private queueCheckInterval = 1000; // 队列检查间隔（毫秒）
  
  // 启动判题守护进程
  async start() {
    console.log('判题守护进程启动，最大并发数：', this.maxConcurrent);
    
    // 定期检查队列
    setInterval(async () => {
      await this.processQueue();
    }, this.queueCheckInterval);
    
    // 定期检查资源
    setInterval(async () => {
      await this.checkResources();
    }, 5000);
  }
  
  // 处理判题队列
  async processQueue() {
    if (this.runningCount >= this.maxConcurrent) {
      return; // 已达到最大并发，等待
    }
    
    // 获取下一个任务（优先级排序）
    const nextTask = await query(`
      SELECT 
        queue_id, solution_id, priority, estimated_time
      FROM judge_queue
      WHERE status = 0
      ORDER BY priority DESC, create_time ASC
      LIMIT 1
    `);
    
    if (nextTask && nextTask.length > 0) {
      await this.judgeTask(nextTask[0]);
    }
  }
  
  // 执行判题任务
  async judgeTask(task: any) {
    this.runningCount++;
    
    try {
      // 更新任务状态
      await query(`
        UPDATE judge_queue
        SET status = 1, start_time = NOW()
        WHERE queue_id = ?
      `, [task.queue_id]);
      
      // 调用判题核心
      const result = await this.doJudge(task.solution_id);
      
      // 更新结果
      await query(`
        UPDATE solution
        SET result = ?, time = ?, memory = ?
        WHERE solution_id = ?
      `, [result.result, result.time, result.memory, task.solution_id]);
      
      // 更新队列状态
      await query(`
        UPDATE judge_queue
        SET status = 2, end_time = NOW()
        WHERE queue_id = ?
      `, [task.queue_id]);
      
    } catch (error) {
      // 错误处理
      await query(`
        UPDATE judge_queue
        SET status = 3, end_time = NOW()
        WHERE queue_id = ?
      `, [task.queue_id]);
    }
    
    this.runningCount--;
  }
  
  // 资源检查
  async checkResources() {
    const cpuUsage = await this.getCPUUsage();
    const memoryUsage = await this.getMemoryUsage();
    
    // 资源超限警告
    if (cpuUsage > 80 || memoryUsage > 1500) {
      console.warn('资源使用超限:', {
        cpu: cpuUsage + '%',
        memory: memoryUsage + 'MB'
      });
      
      // 暂停接受新任务
      if (cpuUsage > 90) {
        this.maxConcurrent = 1; // 降低并发数
      }
    } else {
      this.maxConcurrent = 2; // 恢复并发数
    }
  }
}
```

### 3. **提交排队API**

#### 提交代码API（加入排队）

```typescript
// 提交代码API - 加入判题队列
export async function POST(request: NextRequest) {
  const userId = getUserFromCookies(request)?.userId;
  if (!userId) {
    return NextResponse.json({ error: '未登录' }, { status: 401 });
  }
  
  const { problem_id, language, source } = await request.json();
  
  // 1. 创建提交记录（状态=排队）
  const insertResult = await query(`
    INSERT INTO solution
    (problem_id, user_id, language, result, in_date)
    VALUES (?, ?, ?, 0, NOW())
  `, [problem_id, userId, language]);
  
  const solutionId = insertResult.insertId;
  
  // 2. 获取题目类型（是否比赛题）
  const contest = await query(`
    SELECT contest_id FROM contest_problem
    WHERE problem_id = ?
    ORDER BY contest_id DESC
    LIMIT 1
  `, [problem_id]);
  
  // 3. 计算优先级（比赛题优先）
  const priority = contest.length > 0 ? 10 : 0;
  
  // 4. 预估判题时间
  const estimatedTime = this.estimateJudgeTime(language);
  
  // 5. 加入判题队列
  await query(`
    INSERT INTO judge_queue
    (solution_id, priority, estimated_time, language, create_time)
    VALUES (?, ?, ?, ?, NOW())
  `, [solutionId, priority, estimatedTime, language]);
  
  // 6. 获取当前排队情况
  const queueInfo = await query(`
    SELECT 
      COUNT(*) as total_waiting,
      AVG(CASE WHEN priority >= 10 THEN estimated_time ELSE NULL END) as contest_avg_time,
      AVG(estimated_time) as normal_avg_time
    FROM judge_queue
    WHERE status = 0
  `);
  
  return NextResponse.json({
    success: true,
    solution_id: solutionId,
    queue_info: {
      total_waiting: queueInfo[0].total_waiting,
      estimated_wait: this.calculateWaitTime(queueInfo[0], priority),
      is_contest: priority > 0
    },
    message: '已加入判题队列，请稍候...'
  });
}
```

### 4. **排队状态查询API**

#### 查询排队状态

```typescript
// 查询排队状态API
export async function GET(request: NextRequest) {
  const { searchParams } = new URL(request.url);
  const solutionId = searchParams.get('solution_id');
  
  // 查询队列状态
  const queueStatus = await query(`
    SELECT 
      queue_id, status, priority,
      create_time, start_time, end_time,
      estimated_time,
      TIMESTAMPDIFF(SECOND, create_time, NOW()) as waiting_time
    FROM judge_queue
    WHERE solution_id = ?
  `, [solutionId]);
  
  if (!queueStatus || queueStatus.length === 0) {
    return NextResponse.json({ error: '未找到提交记录' }, { status: 404 });
  }
  
  const status = queueStatus[0];
  
  // 计算预计等待时间
  const estimatedWait = this.calculateRemainingTime(status);
  
  return NextResponse.json({
    solution_id: solutionId,
    status: this.getStatusText(status.status),
    queue_position: await this.getQueuePosition(status.queue_id),
    estimated_wait: estimatedWait,
    waiting_time: status.waiting_time,
    is_judging: status.status === 1
  });
}

// 获取状态文本
getStatusText(status: number) {
  switch (status) {
    case 0: return '排队中';
    case 1: return '判题中';
    case 2: return '已完成';
    case 3: return '判题超时';
    default: return '未知';
  }
}

// 计算队列位置
async getQueuePosition(queueId: number) {
  const position = await query(`
    SELECT COUNT(*) as position
    FROM judge_queue
    WHERE status = 0 AND priority >= (
      SELECT priority FROM judge_queue WHERE queue_id = ?
    )
    AND create_time <= (
      SELECT create_time FROM judge_queue WHERE queue_id = ?
    )
  `, [queueId, queueId]);
  
  return position[0].position;
}
```

### 5. **前端排队状态展示**

#### 实时排队状态组件

```typescript
// 排队状态组件
import { useState, useEffect } from 'react';
import ProgressBar from 'react-bootstrap/ProgressBar';

export default function QueueStatus({ solutionId }) {
  const [queueInfo, setQueueInfo] = useState(null);
  const [progress, setProgress] = useState(0);
  
  useEffect(() => {
    // 定期查询排队状态
    const interval = setInterval(async () => {
      const response = await fetch(`/api/judge/status?solution_id=${solutionId}`);
      const data = await response.json();
      
      setQueueInfo(data);
      
      // 计算进度
      if (data.status === '排队中') {
        const maxWait = data.estimated_wait || 60;
        const current = data.waiting_time || 0;
        setProgress(Math.min(current / maxWait * 100, 100));
      } else if (data.status === '判题中') {
        setProgress(80);
      } else {
        setProgress(100);
        clearInterval(interval); // 完成，停止查询
      }
    }, 2000);
    
    return () => clearInterval(interval);
  }, [solutionId]);
  
  return (
    <div className="queue-status">
      <h5>判题状态</h5>
      <ProgressBar 
        now={progress} 
        label={queueInfo?.status || '查询中'}
        variant={queueInfo?.status === '排队中' ? 'info' : 'success'}
      />
      {queueInfo && (
        <div className="mt-2">
          <p>队列位置：第 {queueInfo.queue_position} 位</p>
          <p>预计等待：{queueInfo.estimated_wait} 秒</p>
          {queueInfo.is_contest && (
            <span className="badge bg-warning">比赛题目优先处理</span>
          )}
        </div>
      )}
    </div>
  );
}
```

## 🎯 优先级策略

### 优先级设计

| 优先级 | 类型 | 处理顺序 |
|--------|------|---------|
| **10** | 正在进行比赛题目 | 最高优先级 |
| **5** | 即将开始的比赛题目 | 高优先级 |
| **0** | 普通练习题目 | 正常优先级 |
| **-5** | 重判题目 | 低优先级 |

### 优先级提升逻辑

```typescript
// 计算题目优先级
async calculatePriority(problemId: number) {
  // 检查是否正在进行比赛
  const activeContest = await query(`
    SELECT contest_id FROM contest
    WHERE start_time <= NOW()
    AND end_time >= NOW()
    AND problem_id IN (
      SELECT problem_id FROM contest_problem WHERE contest_id = contest.contest_id
    )
  `, [problemId]);
  
  if (activeContest.length > 0) {
    return 10; // 最高优先级
  }
  
  // 检查是否即将开始的比赛
  const upcomingContest = await query(`
    SELECT contest_id FROM contest
    WHERE start_time <= NOW() + INTERVAL '30 minutes'
    AND end_time >= NOW()
  `);
  
  if (upcomingContest.length > 0) {
    return 5; // 高优先级
  }
  
  return 0; // 普通优先级
}
```

```typescript
// 超时检查（每10秒）
async checkTimeout() {
  const timeoutTasks = await query(`
    SELECT queue_id, solution_id
    FROM judge_queue
    WHERE status = 1
    AND TIMESTAMPDIFF(SECOND, start_time, NOW()) > 60
  `);
  
  // 标记超时任务
  for (const task of timeoutTasks) {
    await query(`
      UPDATE judge_queue
      SET status = 3, end_time = NOW()
      WHERE queue_id = ?
    `, [task.queue_id]);
    
    await query(`
      UPDATE solution
      SET result = 14, time = 60
      WHERE solution_id = ?
    `, [task.solution_id]); // result=14表示判题超时
    
    this.runningCount--;
  }
}
```

### 判题统计API

```typescript
// 判题统计API
export async function GET(request: NextRequest) {
  const stats = await query(`
    SELECT 
      COUNT(CASE WHEN status = 0 THEN 1 END) as waiting_count,
      COUNT(CASE WHEN status = 1 THEN 1 END) as running_count,
      COUNT(CASE WHEN status = 2 THEN 1 END) as completed_count,
      AVG(CASE WHEN status = 2 THEN
        TIMESTAMPDIFF(SECOND, start_time, end_time)
      END) as avg_judge_time,
      AVG(CASE WHEN status = 0 THEN
        TIMESTAMPDIFF(SECOND, create_time, NOW())
      END) as avg_waiting_time
    FROM judge_queue
    WHERE create_time >= NOW() - INTERVAL '1 hour'
  `);
  
  return NextResponse.json({
    current_concurrent: this.runningCount,
    max_concurrent: this.maxConcurrent,
    waiting_count: stats[0].waiting_count,
    running_count: stats[0].running_count,
    avg_judge_time: stats[0].avg_judge_time,
    avg_waiting_time: stats[0].avg_waiting_time,
    system_status: this.runningCount < this.maxConcurrent ? '正常' : '排队'
  });
}
```
